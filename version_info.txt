#UTF-8, archivo de información de versión para PyInstaller. Windows lee estos datos en las
#Propiedades del .exe (pestaña "Detalles") y en el tooltip del Explorador de Archivos.
#
#Se referencia con --version-file en los argumentos manuales de PyInstaller, que ya están
#puestos en configacadexi.json (auto-py-to-exe, "Argumentos adicionales").
#
#Si subes la versión hay que cambiarla en los CUATRO lugares donde aparece, y los cuatro
#tienen que decir exactamente lo mismo: acá (filevers/prodvers y los FileVersion/ProductVersion
#de texto), en empaquetado/instaladores/AcaDeXI.iss (#define MyAppVersion y VersionInfoVersion),
#en los otros tres .iss (AcaDeXI_Docentes, AcaDeXI_SideStepper, AcaDeXI_SingleTeacher, ahí solo
#MyAppVersion) y en src/interfaz/ventana_principal.py (VERSION, que es el número que muestra la
#ventana "Acerca de"). El nombre del .exe solo se toca en configacadexi.json.

VSVersionInfo(
  ffi=FixedFileInfo(
    #filevers y prodvers van en 4 números: (Mayor, Menor, Parche, Build). Tienen que ser enteros,
    #no texto: si pones "1.0.0" acá el build falla.
    filevers=(1, 0, 0, 0),
    prodvers=(1, 0, 0, 0),
    #Máscara y flags de depuración/estado. 0x3f/0x0 = build de producción, sin flags especiales.
    mask=0x3f,
    flags=0x0,
    #OS objetivo: 0x40004 = VOS_NT_WINDOWS32, o sea un .exe de Windows de 32 o 64 bits.
    OS=0x40004,
    #Tipo de archivo: 0x1 = VFT_APP, una aplicación ejecutable normal.
    fileType=0x1,
    subtype=0x0,
    date=(0, 0)
  ),
  kids=[
    StringFileInfo(
      [
      StringTable(
        u'040904B0',
        [StringStruct(u'CompanyName', u'Omega Slash'),
        StringStruct(u'FileDescription', u'AcaDeXI — Sistema de Gestión Escolar'),
        StringStruct(u'FileVersion', u'1.0.0.0'),
        StringStruct(u'InternalName', u'AcaDeXI'),
        StringStruct(u'LegalCopyright', u'\xa9 2026 Omega Slash. Todos los derechos reservados.'),
        StringStruct(u'OriginalFilename', u'AcaDeXI.exe'),
        StringStruct(u'ProductName', u'AcaDeXI'),
        StringStruct(u'ProductVersion', u'1.0.0.0')])
      ]),
    VarFileInfo([VarStruct(u'Translation', [1033, 1200])])
  ]
)
